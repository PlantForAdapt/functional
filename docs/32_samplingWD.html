Sampling Wood Cores

Some thinking regarding the sampling of wood cores in Belterra, complementing 2025 sampling

Code 
inventory <- readxl::read_excel("data/raw/inventory/inventory_Belterra_20260618.xlsx",
                                sheet = "22092026") %>% 
              select(PLACA, NOME_COMUM, NOME_CIENTIFICO, FAMILIA, 
                     CAP1, CAP2, CAP3, CAP4, CAP5, CAP6, CAP7, E, N) %>% 
              mutate(across(num_range("CAP", 1:7), as.numeric)) %>% 
              rename(tree = PLACA, 
                     vernacular = NOME_COMUM,
                     botanical = NOME_CIENTIFICO,
                     family = FAMILIA,
                     x_utm = E,
                     y_utm = N) %>% 
              mutate(botanical = str_extract(botanical, "^\\S+\\s+\\S+"), 
                     bota_code = paste0(
                       str_sub(word(botanical, 1), 1, 2),
                       str_sub(word(botanical, 2), 1, 2)),
                     label = paste0(bota_code, "_", tree),
                     botanical = replace_values(botanical, "Byrsonima crassifólia" ~ "Byrsonima crassifolia"),
                     bota_code = replace_values(bota_code, "NANA" ~ NA)) %>% 
              rowwise() %>% 
              mutate(dbh = max(CAP1, CAP2, CAP3, CAP4, CAP5, CAP6, CAP7)/pi) %>% 
              select(tree, x_utm, y_utm, 
                     vernacular, botanical, family, bota_code, label, dbh) %>% 
              ungroup() # to remove row_wise


meas_2025 <- read_csv("./outputs/trait_PlantForAdapt_Belterra_2025_v1.csv") %>% 
              mutate(botanical = str_extract(botanical, "^\\S+\\s+\\S+")) %>% 
              group_by(tree, botanical, verna_code) %>% 
              summarise(n_la_2025 = sum(!is.na(la)),
                        n_ldmc_2025 = sum(!is.na(ldmc)),
                        n_lt_2025  = sum(!is.na(lt)),
                        n_sla_2025 = sum(!is.na(sla)),
                        n_tlp_2025 = sum(!is.na(tlp))) %>% 
              ungroup()

wd2025 <- readxl::read_excel("data/raw/Belterra_2025/sample_taken.xlsx",
                                sheet = "Feuil1",
                             na = c("", "NA")) %>%
  rename(tree = ID,
         verna_code = Species) %>% 
  mutate(Wood1 = as.integer(Wood1),
         Wood2 = as.integer(Wood2)) %>% 
  rowwise() %>%
  mutate(n_core_2025 = sum(c(Wood1,Wood2), na.rm=T)) %>% 
  ungroup()


meas_2025 <- meas_2025 %>%
